def robust_z(x, base, min_scale=None):
    med = np.median(base)
    mad = np.median(np.abs(base - med)) * 1.4826
    if min_scale is None:
        min_scale = 0.05 * abs(med) if med != 0 else 1e-3
    return (x - med) / max(mad, min_scale)

def cusum_two_sided(z, k=0.5, h=5):
    s_pos, s_neg = 0.0, 0.0
    alerts = []
    for v in z:
        s_pos = max(0.0, s_pos + v - k)
        s_neg = max(0.0, s_neg - v - k)
        fired = (s_pos > h) or (s_neg > h)
        alerts.append(fired)
        if fired:
            s_pos, s_neg = 0.0, 0.0
    return alerts
